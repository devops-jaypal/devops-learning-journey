CREATE A SERVER
1.name - student-management
2.ami - ubuntu
3.instance type - c71-flex.large
4.storage - 10GB
5.launch instance

ACCESS THE INSTANCE
1.git clone https://github.com/mayurpandit25/student-management.git

DATABASE
1.apt install mysql-server -y
2.CREATE USER 'admin'@'localhost' IDENTIFIED BY 'admin123';
3.GRANT ALL PRIVILEGES ON *.* TO 'admin'@'localhost';
4.FLUSH PRIVILEGES;
5.create database school_db;
6.use school_db;
7.CREATE TABLE IF NOT EXISTS students (
    id     INT AUTO_INCREMENT PRIMARY KEY,
    name   VARCHAR(100)  NOT NULL,
    email  VARCHAR(100)  NOT NULL,
    course VARCHAR(100)  NOT NULL,
    marks  DECIMAL(5,2)  NOT NULL DEFAULT 0
  );

INSTALLATION
1.apt update
2.apt install openjdk-21-jdk -y
3.apt install maven -y
4.java --version
5.mvn --version

APPLICATION
1.cd src/main/resources/
2.vim db.properties
  1.endpoint - 127.0.0.1
  2.username - admin
  3.password - admin123

3.mvn clean package
4.ls target/
  student-management.war

INSTALLATION OF TOMCAT
1.wget https://dlcdn.apache.org/tomcat/tomcat-10/v10.1.61/bin/apache-tomcat-10.1.61.tar.gz
2.tar -xzvf apache-tomcat-10.1.61.tar.gz
3.rm -rf apache-tomcat-10.1.61.tar.gz

COPY WAR FROM STUDENT-MANAGEMENT TO TOMCAT
1.cp -rf student-management/target/student-management.war apache-tomcat-10.1.61/webapps/student.war
2.cd apache-tomcat-10.1.61/bin
3. ./Catalina.sh start

TEST
http://public-ip:8080/student

TOMCAT
1.bin - tomcat star and stop
2.logs - tomcat logs
3.lib - tomcat library files
4.webapps - web application
