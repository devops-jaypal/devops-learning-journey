ADDING THE PYTHON CODE

1.add the code in lambda_function.py

import json
import os
import uuid
import boto3
from decimal import Decimal

dynamodb = boto3.resource("dynamodb")
table = dynamodb.Table(os.environ["STUDENTS_TABLE"])


def convert(value):
    if isinstance(value, Decimal):
        return float(value)
    raise TypeError(f"Cannot serialize {type(value).__name__}")


def reply(status, data):
    return {
        "statusCode": status,
        "headers": {
            "Content-Type": "application/json"
        },
        "body": json.dumps(data, default=convert)
    }


def lambda_handler(event, context):
    method = (
        event.get("requestContext", {})
        .get("http", {})
        .get("method", "GET")
    )

    path = event.get("rawPath", "/").rstrip("/") or "/"

    # Remove the API stage prefix, such as /default.
    stage = event.get("requestContext", {}).get("stage", "")

    if stage and stage != "$default":
        prefix = f"/{stage}"
        if path == prefix:
            path = "/"
        elif path.startswith(prefix + "/"):
            path = path[len(prefix):]

    # POST /students: create a student record.
    if method == "POST" and path == "/students":
        try:
            data = json.loads(event.get("body") or "{}")
        except (json.JSONDecodeError, TypeError):
            return reply(400, {"message": "Invalid JSON"})

        if not isinstance(data, dict):
            return reply(400, {
                "message": "JSON body must be an object"
            })

        required = [
            "name", "rollNumber", "mobile",
            "percentage", "education", "branch"
        ]

        if any(
            data.get(key) is None or not str(data.get(key)).strip()
            for key in required
        ):
            return reply(400, {
                "message": "All student fields are required"
            })

        try:
            percentage = Decimal(str(data["percentage"]))
        except Exception:
            return reply(400, {
                "message": "Percentage must be a number"
            })

        if (
            not percentage.is_finite()
            or not Decimal("0") <= percentage <= Decimal("100")
        ):
            return reply(400, {
                "message": "Percentage must be between 0 and 100"
            })

        student = {
            "studentId": str(uuid.uuid4()),
            "name": str(data["name"]).strip(),
            "rollNumber": str(data["rollNumber"]).strip(),
            "mobile": str(data["mobile"]).strip(),
            "percentage": percentage,
            "education": str(data["education"]).strip(),
            "branch": str(data["branch"]).strip()
        }

        try:
            table.put_item(Item=student)
        except Exception:
            return reply(500, {
                "message": "Could not save student. Check Lambda permissions and DynamoDB."
            })

        return reply(201, {
            "message": "Student saved successfully",
            "student": student
        })

    # GET /students: list student records.
    if method == "GET" and path == "/students":
        try:
            result = table.scan()
            students = result.get("Items", [])

            while "LastEvaluatedKey" in result:
                result = table.scan(
                    ExclusiveStartKey=result["LastEvaluatedKey"]
                )
                students.extend(result.get("Items", []))

            return reply(200, {
                "count": len(students),
                "students": students
            })
        except Exception:
            return reply(500, {
                "message": "Could not retrieve students. Check Lambda permissions and DynamoDB."
            })

    return reply(404, {
        "message": "Route not found",
        "method": method,
        "path": path
    })

    const API_URL = "https://lfnsuyjgq1.execute-api.ap-south-1.amazonaws.com/default/students";

    const form = document.getElementById("studentForm");
    const saveButton = document.getElementById("saveButton");
    const refreshButton = document.getElementById("refreshButton");
    const rows = document.getElementById("studentRows");
    const totalStudents = document.getElementById("totalStudents");
    const apiStatus = document.getElementById("apiStatus");
    const recordCount = document.getElementById("recordCount");

    function showNotice(elementId, message, type) {
      const el = document.getElementById(elementId);
      el.textContent = message;
      el.className = `notice show ${type}`;
    }

    function clearNotice(elementId) {
      const el = document.getElementById(elementId);
      el.textContent = "";
      el.className = "notice";
    }

    function makeCell(value) {
      const td = document.createElement("td");
      // textContent prevents API data from being interpreted as HTML.
      td.textContent = value === undefined || value === null ? "" : String(value);
      return td;
    }

    async function loadStudents() {
      refreshButton.disabled = true;
      apiStatus.textContent = "Connecting…";
      clearNotice("tableNotice");
      try {
        const response = await fetch(API_URL, { method: "GET" });
        const result = await response.json();
        if (!response.ok) throw new Error(result.message || `Request failed (${response.status})`);

        const students = Array.isArray(result.students) ? result.students : [];
        rows.replaceChildren();

        if (students.length === 0) {
          const tr = document.createElement("tr");
          const td = document.createElement("td");
          td.colSpan = 6;
          td.className = "empty";
          td.textContent = "No students yet. Add the first student using the form.";
          tr.appendChild(td);
          rows.appendChild(tr);
        } else {
          students.forEach(student => {
            const tr = document.createElement("tr");
            [student.name, student.rollNumber, student.mobile,
             student.percentage, student.education, student.branch]
              .forEach(value => tr.appendChild(makeCell(value)));
            rows.appendChild(tr);
          });
        }

        totalStudents.textContent = String(students.length);
        recordCount.textContent = `${students.length} record${students.length === 1 ? "" : "s"} found`;
        apiStatus.textContent = "Connected";
        apiStatus.style.color = "var(--green)";
      } catch (error) {
        rows.replaceChildren();
        const tr = document.createElement("tr");
        const td = document.createElement("td");
        td.colSpan = 6;
        td.className = "empty";
        td.textContent = "Could not load records. Check API Gateway CORS settings and Lambda logs.";
        tr.appendChild(td);
        rows.appendChild(tr);
        totalStudents.textContent = "—";
        recordCount.textContent = "Unable to load records";
        apiStatus.textContent = "Connection error";
        apiStatus.style.color = "var(--red)";
        showNotice("tableNotice", error.message || "API request failed. Check CORS configuration.", "error");
      } finally {
        refreshButton.disabled = false;
      }
    }

    form.addEventListener("submit", async event => {
      event.preventDefault();
      clearNotice("formNotice");
      saveButton.disabled = true;
      saveButton.textContent = "Saving…";

      const payload = {
        name: form.name.value.trim(),
        rollNumber: form.rollNumber.value.trim(),
        mobile: form.mobile.value.trim(),
        percentage: Number(form.percentage.value),
        education: form.education.value.trim(),
        branch: form.branch.value.trim()
      };

      try {
        const response = await fetch(API_URL, {
          method: "POST",
          headers: { "Content-Type": "application/json" },
          body: JSON.stringify(payload)
        });
        const result = await response.json();
        if (!response.ok) throw new Error(result.message || `Save failed (${response.status})`);

        showNotice("formNotice", "Student saved successfully.", "success");
        form.reset();
        await loadStudents();
      } catch (error) {
        showNotice(
          "formNotice",
          `${error.message || "Could not save student."} If the browser reports a CORS error, configure API Gateway CORS to allow GET, POST, OPTIONS and the content-type header.`,
          "error"
        );
      } finally {
        saveButton.disabled = false;
        saveButton.textContent = "Save student";
      }
    });

    refreshButton.addEventListener("click", loadStudents);
    form.addEventListener("reset", () => clearNotice("formNotice"));
    loadStudents();
  </script>
